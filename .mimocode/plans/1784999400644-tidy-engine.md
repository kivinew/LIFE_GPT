```python
def disease_phase(self, cells, grid):
    if self.sick:
        self.sick_timer -= 1
        if self.sick_timer <= 0:
            self.sick = False
        else:
            self.energy -= 0.1 * dt  # потеря энергии от болезни
    elif random.random() < DISEASE_CHANCE:
        # Проверить, есть ли больные рядом
        for j in get_neighbors(grid, self.x, self.y, radius=1):
            other = cells[j]
            if other is not self and other.sick:
                dist = math.hypot(other.x - self.x, other.y - self.y)
                if dist < DISEASE_TRANSMISSION_RANGE:
                    self.sick = True
                    self.sick_timer = DISEASE_DURATION
                    break
```

**Файл**: `cell.py` — в `metabolism_phase`:
```python
def metabolism_phase(self, dt):
    d = self.genome.diet
    metab_mult = BASE_METABOLISM_MULT if d == PHOT else PREDATOR_METABOLISM_MULT
    if self.sick:
        metab_mult *= DISEASE_METABOLISM_MULT
    self.energy -= (
        self.genome.metabolism +
        self.genome.speed * SPEED_COST +
        self.genome.mass * self.genome.mass * MASS_PENALTY
    ) * metab_mult * dt
```

```python
    if random.random() < MIGRATION_CHANCE:
        if self.energy < self.max_energy * 0.3:  # мигрируют только голодные
            angle = random.random() * math.tau
            dist = random.random() * MIGRATION_DISTANCE
            new_x = self.x + math.cos(angle) * dist
            new_y = self.y + math.sin(angle) * dist
            # Не выходить за границы
            self.x = max(0, min(W - SB - 1, new_x))
```

```python
def test_diet_balance():
    """Все три диеты должны сосуществовать."""
    cells = []
    for _ in range(40):
        cells.append(Cell(random.uniform(100, 500), random.uniform(100, 500), 
                         Genome(diet=PHOT)))
    for _ in range(30):
        cells.append(Cell(random.uniform(100, 500), random.uniform(100, 500), 
                         Genome(diet=ZOOP)))
    for _ in range(30):
        cells.append(Cell(random.uniform(100, 500), random.uniform(100, 500), 
                         Genome(diet=POLY)))
    
    field = ResourceField(1000, 800)
    for tick in range(10000):
        grid = build_spatial_grid(cells)
        for c in cells:
            c.step(field, cells, grid, 1.0, tick)
        cells = [c for c in cells if not c.is_dead()]
    
    # После 10000 тиков все три диеты должны выжить
    phot = sum(1 for c in cells if c.genome.diet == PHOT)
    zoo = sum(1 for c in cells if c.genome.diet == ZOOP)
    poly = sum(1 for c in cells if c.genome.diet == POLY)
    
    assert phot > 0, "PHOT вымерли"
    assert zoo > 0, "ZOOP вымерли"
    assert poly > 0, "POLY вымерли"
```
